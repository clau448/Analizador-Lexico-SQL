cmake_minimum_required(VERSION 3.16)
project(AnalizadorSQL LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -pedantic)
endif()

add_library(sql_lexer Lexer.cpp)
target_include_directories(sql_lexer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(lexer main.cpp)
target_link_libraries(lexer PRIVATE sql_lexer)

add_library(sql_ll1 parser_ll1/Grammar.cpp parser_ll1/Parser.cpp)
target_link_libraries(sql_ll1 PUBLIC sql_lexer)
add_executable(sql_parser parser_ll1/main.cpp)
target_link_libraries(sql_parser PRIVATE sql_ll1)

include(CTest)
if(BUILD_TESTING)
  foreach(test_name IN ITEMS grammar_tests parser_tests validation_tests)
    add_executable(${test_name} parser_ll1/tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE sql_ll1)
    # These suites use assert; keep checks enabled in Release as well.
    if(MSVC)
      target_compile_options(${test_name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  add_test(NAME lexer_demo COMMAND lexer)
  add_test(NAME parser_cli_valid COMMAND sql_parser "SELECT nombre FROM alumnos WHERE edad >= 18;")
  set_tests_properties(parser_cli_valid PROPERTIES PASS_REGULAR_EXPRESSION "ACEPTADA")
  add_test(NAME parser_cli_trace COMMAND sql_parser --trace "SELECT * FROM alumnos;")
  set_tests_properties(parser_cli_trace PROPERTIES PASS_REGULAR_EXPRESSION "match \\$" )
  add_test(NAME parser_cli_invalid COMMAND sql_parser "SELECT FROM alumnos;")
  set_tests_properties(parser_cli_invalid PROPERTIES WILL_FAIL TRUE)
endif()
